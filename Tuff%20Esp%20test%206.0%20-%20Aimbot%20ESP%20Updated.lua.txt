local OpenCloseKey = Enum.KeyCode.Insert

local WaitingForKey = false

local function LoadSavedKey()

if not isfile or not readfile then

    return

end



local Exists = false



pcall(function()

    Exists = isfile(KeybindFile)

end)



if not Exists then

    return

end



local Success, SavedKey = pcall(function()

    return readfile(KeybindFile)

end)



if not Success or not SavedKey or SavedKey == "" then

    return

end



SavedKey = SavedKey:gsub("%s+", "")



local Success2, KeyCodeValue = pcall(function()

    return Enum.KeyCode[SavedKey]

end)



if Success2 and KeyCodeValue then

    OpenCloseKey = KeyCodeValue

end

end

local function SaveKeybind()

if not writefile then

    return

end



if makefolder and isfolder then

    pcall(function()

        if not isfolder(SaveFolder) then

            makefolder(SaveFolder)

        end

    end)

end



pcall(function()

    writefile(KeybindFile, OpenCloseKey.Name)

end)

end

LoadSavedKey()

local AutoLoadEnabled = false

local TuffSkinChangerURL =

"https://raw.githubusercontent.com/Dailychomik127/TuffSkinChanger/refs/heads/main/TuffSkinChanger.lua"

local function LoadAutoLoadState()

if not isfile or not readfile then

    return

end



local Exists = false



pcall(function()

    Exists = isfile(AutoLoadFile)

end)



if not Exists then

    AutoLoadEnabled = false

    return

end



local Success, State = pcall(function()

    return readfile(AutoLoadFile)

end)



if not Success or not State then

    AutoLoadEnabled = false

    return

end



State = State:gsub("%s+", ""):lower()

AutoLoadEnabled = State == "on"

end

local function SaveAutoLoadState()

if not writefile then

    return

end



if makefolder and isfolder then

    pcall(function()

        if not isfolder(SaveFolder) then

            makefolder(SaveFolder)

        end

    end)

end



pcall(function()

    writefile(

        AutoLoadFile,

        AutoLoadEnabled and "on" or "off"

    )

end)

end

LoadAutoLoadState()

    HitboxBodySmall = true,
    HitboxHead = true,

    PhysicalHitboxHead = true,

--//==================================================
--// REMOVE ESP
--//==================================================

local function RemoveESP(character)
    if not character then
        return
    end

    for _, obj in ipairs(character:GetDescendants()) do
        if obj:IsA("BoxHandleAdornment") and obj.Name == "ESPBox" then
            obj:Destroy()
        end
    end
end

--//==================================================
--// ESP PART FILTER
--//==================================================

local function ShouldESPPart(part)
    if not part:IsA("BasePart") then
        return false
    end

    if ESPExcludedParts[part.Name] then
        return false
    end

    if EspPartMode == "All" then
        return true
    elseif EspPartMode == "Head" then
        return part.Name == "Head"
    elseif EspPartMode == "Torso" then
        return part.Name == "LowerTorso" or part.Name == "UpperTorso"
    end

    return false
end

--//==================================================
--// APPLY ESP
--//==================================================

local function ApplyESP(character)
    if not EspEnabled or not character then
        return
    end

    RemoveESP(character)

    for _, part in ipairs(character:GetDescendants()) do
        if ShouldESPPart(part) then
            local box = Instance.new("BoxHandleAdornment")
            box.Name = "ESPBox"
            box.Size = part.Size
            box.Adornee = part
            box.Color3 = Color3.new(1, 0, 0)
            box.AlwaysOnTop = true
            box.ZIndex = 10
            box.Transparency = 0.5
            box.Parent = character
        end
    end
end

--//==================================================
--// APPLY ESP TO EVERY PLAYER

local function ApplyESPToAll()
    if not EspEnabled then
        return
    end

    for _, player in ipairs(Players:GetPlayers()) do

local function ToggleGUI()

if BlurTween then

    BlurTween:Cancel()

    BlurTween = nil

end



if ScreenGui.Enabled then

    ScreenGui.Enabled = false

    Blur.Size = 8



    BlurTween = TweenService:Create(

        Blur,

        BlurTweenInfo,

        {

            Size = 0

        }

    )



    BlurTween:Play()

else

    ScreenGui.Enabled = true

    Blur.Size = 0



    BlurTween = TweenService:Create(

        Blur,

        BlurTweenInfo,

        {

            Size = 8

        }

    )



    BlurTween:Play()

end

end

--// MOBILE FLOATING BUTTON
--// Only appears on mobile devices.
--// The button stays visible while the main GUI is hidden.
--// It can be dragged anywhere on the screen and tapped to toggle the GUI.

--//==================================================

local MobileButtonGui
local MobileButton

if IsMobile then

    MobileButtonGui = Instance.new("ScreenGui")
    MobileButtonGui.Name = "TuffMobileButtonGUI"
    MobileButtonGui.ResetOnSpawn = false
    MobileButtonGui.IgnoreGuiInset = true
    MobileButtonGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling

    if syn and syn.protect_gui then
        syn.protect_gui(MobileButtonGui)
    end

    MobileButtonGui.Parent = GuiParent

    MobileButton = Instance.new("TextButton")
    MobileButton.Name = "MobileFloatingButton"
    MobileButton.Size = UDim2.new(0, 58, 0, 58)
    MobileButton.Position = UDim2.new(1, -78, 1, -145)
    MobileButton.BackgroundColor3 = Color3.fromRGB(18, 42, 70)
    MobileButton.BackgroundTransparency = 0.08
    MobileButton.BorderSizePixel = 0
    MobileButton.AutoButtonColor = false
    MobileButton.Text = "T"
    MobileButton.TextColor3 = Color3.fromRGB(230, 240, 255)
    MobileButton.TextSize = 24
    MobileButton.Font = Enum.Font.GothamBold
    MobileButton.Active = true
    MobileButton.ZIndex = 100
    MobileButton.Parent = MobileButtonGui

    local MobileButtonCorner = Instance.new("UICorner")
    MobileButtonCorner.CornerRadius = UDim.new(0, 18)
    MobileButtonCorner.Parent = MobileButton

    local MobileButtonStroke = Instance.new("UIStroke")
    MobileButtonStroke.Color = Color3.fromRGB(65, 105, 155)
    MobileButtonStroke.Transparency = 0.25
    MobileButtonStroke.Thickness = 1.5
    MobileButtonStroke.Parent = MobileButton

    local MobileButtonDragging = false
    local MobileButtonDragStart = nil
    local MobileButtonStartPosition = nil
    local MobileButtonDragInput = nil
    local MobileButtonMoved = false

    local function UpdateMobileButtonDrag(Input)
        if not MobileButtonDragStart or not MobileButtonStartPosition then
            return
        end

        local Delta = Input.Position - MobileButtonDragStart

        if math.abs(Delta.X) > 6 or math.abs(Delta.Y) > 6 then
            MobileButtonMoved = true
        end

        MobileButton.Position = UDim2.new(
            MobileButtonStartPosition.X.Scale,
            MobileButtonStartPosition.X.Offset + Delta.X,
            MobileButtonStartPosition.Y.Scale,
            MobileButtonStartPosition.Y.Offset + Delta.Y
        )
    end

    MobileButton.InputBegan:Connect(function(Input)
        if Input.UserInputType == Enum.UserInputType.Touch
            or Input.UserInputType == Enum.UserInputType.MouseButton1 then

            MobileButtonDragging = true
            MobileButtonMoved = false
            MobileButtonDragStart = Input.Position
            MobileButtonStartPosition = MobileButton.Position
        end
    end)

    MobileButton.InputChanged:Connect(function(Input)
        if Input.UserInputType == Enum.UserInputType.Touch
            or Input.UserInputType == Enum.UserInputType.MouseMovement then

            MobileButtonDragInput = Input
        end
    end)

    UserInputService.InputChanged:Connect(function(Input)
        if Input == MobileButtonDragInput and MobileButtonDragging then
            UpdateMobileButtonDrag(Input)
        end
    end)

    UserInputService.InputEnded:Connect(function(Input)
        if Input.UserInputType == Enum.UserInputType.Touch
            or Input.UserInputType == Enum.UserInputType.MouseButton1 then

            if MobileButtonDragging then
                MobileButtonDragging = false

                if not MobileButtonMoved then
                    ToggleGUI()
                end
            end
        end
    end)

    MobileButton.MouseEnter:Connect(function()
        MobileButton.BackgroundColor3 = Color3.fromRGB(25, 52, 85)
    end)

    MobileButton.MouseLeave:Connect(function()
        MobileButton.BackgroundColor3 = Color3.fromRGB(18, 42, 70)
    end)
end

local Dragging = false

local DragStart = nil

local StartPosition = nil

local DragInput = nil

local function UpdateDrag(Input)

local Delta = Input.Position - DragStart



Main.Position = UDim2.new(

    StartPosition.X.Scale,

    StartPosition.X.Offset + Delta.X,

    StartPosition.Y.Scale,

    StartPosition.Y.Offset + Delta.Y

)

end

DragBar.InputBegan:Connect(function(Input)

if Input.UserInputType == Enum.UserInputType.MouseButton1

    or Input.UserInputType == Enum.UserInputType.Touch then



    Dragging = true

    DragStart = Input.Position

    StartPosition = Main.Position

end

end)

DragBar.InputChanged:Connect(function(Input)

if Input.UserInputType == Enum.UserInputType.MouseMovement

    or Input.UserInputType == Enum.UserInputType.Touch then



    DragInput = Input

end

end)

UserInputService.InputChanged:Connect(function(Input)

if Input == DragInput and Dragging then

    UpdateDrag(Input)

end

end)

UserInputService.InputEnded:Connect(function(Input)

if Input.UserInputType == Enum.UserInputType.MouseButton1

    or Input.UserInputType == Enum.UserInputType.Touch then



    Dragging = false

end

end)
